+# שדרוג OpenClaw לעברית
+
+הערכה הזו בונה גרסת Windows בעברית ומתקינה אותה כשדרוג מעל ההתקנה הרגילה של OpenClaw Companion.
+
+אין להסיר קודם את האפליקציה האנגלית. המתקין משתמש באותו מזהה אפליקציה ובאותה תיקיית התקנה, ולכן הוא מחליף את קובצי התוכנה ושומר את ההגדרות, חיבורי ה־Gateway ונתוני המשתמש.
+
+## הפעלה
+
+1. חלץ את כל קובצי ה־ZIP לתיקייה רגילה במחשב.
+2. פתח PowerShell בתוך התיקייה שחולצה.
+3. הרץ:
+
+```powershell
+powershell -ExecutionPolicy Bypass -File .\UPGRADE-TO-HEBREW.ps1
+```
+
+הסקריפט יבצע את הפעולות הבאות:
+
+1. יוודא שכלי הבנייה הדרושים מותקנים. אם חסר כלי, הוא יתקין אותו באמצעות winget.
+2. יריץ את הבנייה המלאה ואת בדיקות המאגר.
+3. יבנה מתקין Release עבור win-x64, בלי זהות הפיתוח הנפרדת.
+4. יסגור את OpenClaw רק לאחר שהבנייה והבדיקות הצליחו.
+5. יגבה את `%APPDATA%\OpenClawTray` ואת `%LOCALAPPDATA%\OpenClawTray` לתיקייה במסמכים.
+6. יפעיל את המתקין כשדרוג מעל האפליקציה הקיימת.
+7. יגדיר את `OPENCLAW_LANGUAGE` לערך `he-IL` ויפעיל את האפליקציה.
+
+המחשב צריך להיות מחובר לאינטרנט בזמן הבנייה הראשונה. התהליך עשוי להימשך זמן מה ולהציג בקשות הרשאה של Windows.
+
+## אחרי ההתקנה
+
+בדוק את מסך הפתיחה, Settings, תפריט מגש המערכת, חלונות האישור והחיבור ל־Gateway. אל תמחק את תיקיית הגיבוי עד שהכול עובד.
+
+אם התהליך נעצר, אין להסיר את האפליקציה. העתק את כל הפלט מחלון PowerShell ושלח אותו לבדיקה. ההתקנה הקיימת אינה משתנה לפני שהבנייה, הבדיקות ויצירת המתקין מסתיימות בהצלחה.

+[CmdletBinding()]
+param()
+
+Set-StrictMode -Version Latest
+$ErrorActionPreference = "Stop"
+
+$repoRoot = $PSScriptRoot
+$installerVersion = "2026.9.1"
+$installedExe = Join-Path $env:LOCALAPPDATA "OpenClawTray\OpenClaw.Tray.WinUI.exe"
+$backupRoot = $null
+$previousLanguage = [Environment]::GetEnvironmentVariable("OPENCLAW_LANGUAGE", "User")
+$languageChanged = $false
+$installSucceeded = $false

+function Write-Step {
+    param([string]$Message)
+    Write-Host "`n=== $Message ===" -ForegroundColor Cyan
+}
+
+function Refresh-ProcessPath {
+    $machinePath = [Environment]::GetEnvironmentVariable("Path", "Machine")
+    $userPath = [Environment]::GetEnvironmentVariable("Path", "User")
+    $env:Path = @($machinePath, $userPath) -join ";"
+}
+
+function Invoke-WingetInstall {
+    param(
+        [Parameter(Mandatory = $true)]
+        [string]$PackageId
+    )
+
+    if (-not (Get-Command winget.exe -ErrorAction SilentlyContinue)) {
+        throw "Windows Package Manager (winget) is required. Install App Installer from Microsoft Store, then rerun this script."
+    }
+
+    Write-Step "Installing prerequisite: $PackageId"
+    & winget.exe install --id $PackageId -e --accept-source-agreements --accept-package-agreements --disable-interactivity
+    if ($LASTEXITCODE -ne 0) {
+        throw "winget could not install $PackageId (exit code $LASTEXITCODE)."
+    }
+
+    Refresh-ProcessPath
+}
+
+function Ensure-Prerequisites {
+    Write-Step "Checking Windows build prerequisites"
+
+    if (-not [System.Environment]::Is64BitOperatingSystem) {
+        throw "This upgrade kit requires 64-bit Windows."
+    }
+
+    if (-not (Get-Command git.exe -ErrorAction SilentlyContinue)) {
+        Invoke-WingetInstall -PackageId "Git.Git"
+    }
+
+    $hasNet10 = $false
+    if (Get-Command dotnet.exe -ErrorAction SilentlyContinue) {
+        $hasNet10 = [bool](& dotnet.exe --list-sdks 2>$null | Where-Object { $_ -match "^10\." })
+    }
+    if (-not $hasNet10) {
+        Invoke-WingetInstall -PackageId "Microsoft.DotNet.SDK.10"
+    }
+
+    if (-not (Get-Command npm.cmd -ErrorAction SilentlyContinue)) {
+        Invoke-WingetInstall -PackageId "OpenJS.NodeJS.LTS"
+    }
+
+    $windowsSdkRoot = Join-Path ${env:ProgramFiles(x86)} "Windows Kits\10\Include"
+    $hasWindowsSdk = Test-Path -LiteralPath $windowsSdkRoot
+    if (-not $hasWindowsSdk) {
+        Invoke-WingetInstall -PackageId "Microsoft.WindowsSDK.10.0.26100"
+    }
+
+    Refresh-ProcessPath
+
+    foreach ($command in @("git.exe", "dotnet.exe", "npm.cmd")) {
+        if (-not (Get-Command $command -ErrorAction SilentlyContinue)) {
+            throw "$command is still unavailable after prerequisite setup. Restart Windows and rerun this script."
+        }
+    }
+}
+
+function Invoke-DotNetTest {
+    param([string]$Project)
+
+    & dotnet.exe test $Project --no-restore
+    if ($LASTEXITCODE -ne 0) {
+        throw "Tests failed: $Project"
+    }
+}
+
+function Stop-OpenClaw {
+    $processes = Get-Process -Name "OpenClaw.Tray.WinUI" -ErrorAction SilentlyContinue
+    if ($processes) {
+        Write-Step "Closing the running OpenClaw app"
+        $processes | Stop-Process -Force
+        Start-Sleep -Seconds 2
+    }
+}
+
+function Copy-DirectoryToBackup {
+    param(
+        [string]$Source,
+        [string]$Destination
+    )
+
+    if (-not (Test-Path -LiteralPath $Source)) {
+        return
+    }
+
+    $parent = Split-Path -Parent $Destination
+    New-Item -ItemType Directory -Path $parent -Force | Out-Null
+    Copy-Item -LiteralPath $Source -Destination $Destination -Recurse -Force
+}
+
+function Backup-OpenClawData {
+    $documents = [Environment]::GetFolderPath([Environment+SpecialFolder]::MyDocuments)
+    $script:backupRoot = Join-Path $documents ("OpenClaw-Hebrew-Backup-" + (Get-Date -Format "yyyyMMdd-HHmmss"))
+    New-Item -ItemType Directory -Path $script:backupRoot -Force | Out-Null
+
+    $roamingSource = Join-Path $env:APPDATA "OpenClawTray"
+    $localSource = Join-Path $env:LOCALAPPDATA "OpenClawTray"
+    Copy-DirectoryToBackup -Source $roamingSource -Destination (Join-Path $script:backupRoot "Roaming\OpenClawTray")
+    Copy-DirectoryToBackup -Source $localSource -Destination (Join-Path $script:backupRoot "Local\OpenClawTray")
+
+    @(
+        "Created: $(Get-Date -Format o)"
+        "Roaming source: $roamingSource"
+        "Local source: $localSource"
+        "Upgrade identity: release"
+        "Upgrade version: $installerVersion"
+    ) | Set-Content -LiteralPath (Join-Path $script:backupRoot "backup-info.txt") -Encoding UTF8
+
+    Write-Host "Backup created at: $script:backupRoot" -ForegroundColor Green
+}
